#!/usr/bin/env bash
# nexus-security-check.sh — quick posture audit
NEXUS_HOME="${NEXUS_HOME:-$HOME/NeXuS}"
LOG="$NEXUS_HOME/logs/security.log"

check_service() {
    local svc="$1"
    if rc-service "$svc" status &>/dev/null; then
        echo "[OK]  $svc running"
    else
        echo "[!!]  $svc DOWN" | tee -a "$LOG"
    fi
}

for svc in psad fail2ban ossec-hids; do
    check_service "$svc"
done

# AIDE integrity check (offline hashes)
aide --check 2>&1 | grep -E "^(Added|Removed|Changed):" | head -20

#!/usr/bin/env python3
"""nexus_metrics_collector.py — node health aggregator"""
import json, subprocess
from pathlib import Path

NEXUS_HOME = Path.home() / "NeXuS"
SERVICES = ["psad", "fail2ban", "ossec-hids", "unbound"]

def check_service(name: str) -> dict:
    result = subprocess.run(
        ["rc-service", name, "status"],
        capture_output=True, text=True
    )
    return {
        "name": name,
        "running": result.returncode == 0,
        "output": result.stdout.strip()
    }

if __name__ == "__main__":
    report = {
        "node": "nexus-node-01",
        "services": [check_service(s) for s in SERVICES],
    }
    print(json.dumps(report, indent=2))

// nexus-web-server — universal API proxy
const NEXUS_HOME = process.env.NEXUS_HOME ?? `${process.env.HOME}/NeXuS`;
const PORT = parseInt(process.env.NWS_PORT ?? '8443');

const providers = Object.fromEntries(
  Object.entries(process.env)
    .filter(([k]) => k.startsWith('PROVIDER_'))
    .map(([k, v]) => [k.replace('PROVIDER_', ''), v])
);

async function handleProxy(req, res) {
  const provider = providers[req.params.id];
  if (!provider) return res.status(404).json({ error: 'unknown provider' });
  // keys stay server-side — never in browser
  const upstream = await fetch(provider.url, {
    headers: { Authorization: `Bearer ${provider.key}` },
    body: req.body
  });
  return res.json(await upstream.json());
}
